\section*{Software Description}
\begin{itemize}[noitemsep,topsep=1pt,leftmargin=1.1em]
    \item \textbf{OS \& Middleware:} Ubuntu~24.04~LTS, ROS~2 Jazzy, Zenoh (\texttt{rmw\_zenoh\_cpp}), Pixi + RoboStack
    \item \textbf{Navigation:} SLAM Toolbox (dual 2D LiDAR + omni odometry); Nav2 (AMCL, MPPI, Rotation Shim, STVL ZED~2i depth, Collision Monitor, keepout zones)
    \item \textbf{Manipulation:} MoveIt~2 with C++ bimanual commander, TRAC-IK, OMPL RRT-Connect, Ruckig; opt-in whole-body planning (base + lift + arm) validated in MuJoCo, under real-robot trial
    \item \textbf{Simulation:} Shared URDF/MJCF across MuJoCo, Gazebo, and Isaac Sim; in-house Blender-to-MuJoCo room exporter
    \item \textbf{Telemetry:} RViz~2, Foxglove Studio (live reasoning logs, checklist, 3D memory and grasp markers)
    \item \textbf{Agent \& LLM:} Single ROS~2 ReAct agent (RAI, LangChain, LangGraph) with safety middleware; local LLM via Ollama or OpenAI-compatible server (model TBD), optional cloud API (tested with DeepSeek~V4.1~Flash via OpenRouter) when venue internet is available % TODO(verify)
    \item \textbf{Vision (Local GPU Node):} YOLOE-26s-seg + ByteTrack (open-vocab objects); YOLO26s-pose + BoT-SORT with CMC (pose/gestures); SCRFD + ArcFace \texttt{w600k\_r50} (ONNX) and OSNet~x1.0 (face/body re-ID); CLIP ViT-B/16 embeddings (SigLIP~2 / EmbeddingGemma~2 bake-off ongoing) % TODO(verify)
    \item \textbf{Memory:} Unified SQLite (WAL) + \texttt{sqlite-vec} node (ConceptGraphs-inspired online 3D object/surface memory; session-scoped ArcFace/OSNet person memory)
    \item \textbf{Speech (Half-Duplex):} Local Silero~VAD + Whisper/FasterWhisper (optional cloud STT); local CPU ONNX Kokoro~TTS; YAMNet doorbell/knock detector (in development) % TODO(verify)
    \item \textbf{Public Repositories \& Docs:} \url{https://github.com/EIC-Robocup-2026} (code, documentation, ADRs)
\end{itemize}
